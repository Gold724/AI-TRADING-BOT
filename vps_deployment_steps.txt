## STEP 2: EXECUTE DEPLOYMENT SCRIPT

### Create and Run the Deployment Script
```bash
# Navigate to root directory
cd /root

# Create the deployment script
cat > deploy_trae_bot.sh << 'EOF'
#!/bin/bash
set -e

echo "=== AI Trading Sentinel - VPS Deployment ==="
echo "Starting deployment on Ubuntu 24.04 at $(date)..."

# Update system
echo "[1/10] Updating system packages..."
apt update && apt upgrade -y

# Install essential packages
echo "[2/10] Installing essential packages..."
apt install -y python3 python3-pip python3-venv nodejs npm git curl wget unzip htop

# Install Python packages
echo "[3/10] Installing Python packages..."
pip3 install --upgrade pip
pip3 install playwright pyyaml requests flask python-dotenv pandas numpy aiohttp websockets gunicorn

# Install Playwright browsers
echo "[4/10] Installing Playwright browsers..."
playwright install
playwright install-deps

# Clone or update repository
echo "[5/10] Setting up bot repository..."
if [ ! -d "ai-trading-sentinel" ]; then
    echo "Cloning repository..."
    git clone https://github.com/YOUR_USERNAME/ai-trading-sentinel.git
else
    echo "Repository exists, pulling latest changes..."
    cd ai-trading-sentinel
    git pull origin main
    cd ..
fi

cd ai-trading-sentinel

# Create logs directory
echo "[6/10] Creating logs directory..."
mkdir -p logs
touch logs/trae.log
touch logs/health.log
chmod 755 logs
chmod 644 logs/*.log

# Fix main.py - Replace selenium with playwright
echo "[7/10] Fixing main.py imports..."
if [ -f "main.py" ]; then
    # Backup original
    cp main.py main.py.backup
    
    # Comment out selenium imports
    sed -i 's/^from selenium/#from selenium/g' main.py
    sed -i 's/^import selenium/#import selenium/g' main.py
    
    # Add playwright imports if not present
    if ! grep -q "from playwright" main.py; then
        echo "" >> main.py
        echo "# Playwright imports (replacing selenium)" >> main.py
        echo "from playwright.async_api import async_playwright" >> main.py
        echo "from playwright.sync_api import sync_playwright" >> main.py
    fi
    
    echo "✅ main.py updated (selenium → playwright)"
fi

# Create systemd service
echo "[8/10] Creating systemd service..."
cat > /etc/systemd/system/trae-bot.service << 'SERVICE_EOF'
[Unit]
Description=AI Trading Sentinel Bot
After=network.target network-online.target
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/root/ai-trading-sentinel
Environment=PYTHONPATH=/root/ai-trading-sentinel
Environment=DISPLAY=:0
Environment=PYTHONUNBUFFERED=1
ExecStartPre=/bin/bash -c 'echo "[$(date)] Starting TRAE Bot..." >> /root/ai-trading-sentinel/logs/trae.log'
ExecStart=/usr/bin/python3 /root/ai-trading-sentinel/main.py
Restart=always
RestartSec=10
StartLimitInterval=300
StartLimitBurst=5
StandardOutput=append:/root/ai-trading-sentinel/logs/trae.log
StandardError=append:/root/ai-trading-sentinel/logs/trae.log

[Install]
WantedBy=multi-user.target
SERVICE_EOF

# Reload systemd and enable service
echo "[9/10] Configuring systemd service..."
systemctl daemon-reload
systemctl enable trae-bot.service

# Test Python imports
echo "[10/10] Testing Python imports..."
python3 -c "import playwright; import yaml; import requests; import flask; print('✅ All critical imports successful!')"

echo ""
echo "=== DEPLOYMENT COMPLETE ==="
echo "✅ System updated and packages installed"
echo "✅ Playwright browsers installed"
echo "✅ Repository cloned/updated"
echo "✅ Logs directory created"
echo "✅ main.py fixed (selenium → playwright)"
echo "✅ Systemd service configured and enabled"
echo "✅ Python imports verified"
echo ""
echo "=== NEXT STEPS ==="
echo "1. Configure .env file: nano /root/ai-trading-sentinel/.env"
echo "2. Start service: systemctl start trae-bot"
echo "3. Check status: systemctl status trae-bot"
echo "4. View logs: tail -f /root/ai-trading-sentinel/logs/trae.log"
echo ""
EOF

# Make script executable
chmod +x deploy_trae_bot.sh

# Execute the deployment script
echo "Executing deployment script..."
./deploy_trae_bot.sh
```

    # Test manual execution
    echo "Testing manual execution..."

    python3 main.py

## STEP 5: VERIFY 24/7 OPERATION

### Setup Monitoring Tools
```bash
# Create health check script
cat > /root/health_check.sh << 'EOF'
#!/bin/bash
LOG_FILE="/root/ai-trading-sentinel/logs/health.log"
DATE=$(date '+%Y-%m-%d %H:%M:%S')

# Check if service is running
if ! systemctl is-active --quiet trae-bot; then
    echo "[$DATE] ALERT: TRAE Bot service is down, attempting restart..." >> $LOG_FILE
    systemctl restart trae-bot
    sleep 30
    
    if systemctl is-active --quiet trae-bot; then
        echo "[$DATE] SUCCESS: TRAE Bot service restarted successfully" >> $LOG_FILE
    else
        echo "[$DATE] CRITICAL: TRAE Bot service failed to restart" >> $LOG_FILE
        # Send alert (implement your notification method)
    fi
else
    echo "[$DATE] OK: TRAE Bot service is running" >> $LOG_FILE
fi

# Check system resources
MEM_USAGE=$(free | grep Mem | awk '{printf "%.1f", $3/$2 * 100.0}')
DISK_USAGE=$(df /root | tail -1 | awk '{print $5}' | sed 's/%//')

echo "[$DATE] System Status - Memory: ${MEM_USAGE}%, Disk: ${DISK_USAGE}%" >> $LOG_FILE

# Alert if resources are high
if (( $(echo "$MEM_USAGE > 90" | bc -l) )); then
    echo "[$DATE] WARNING: High memory usage: ${MEM_USAGE}%" >> $LOG_FILE
fi

if (( DISK_USAGE > 85 )); then
    echo "[$DATE] WARNING: High disk usage: ${DISK_USAGE}%" >> $LOG_FILE
fi
EOF

chmod +x /root/health_check.sh

# Add to crontab (check every 5 minutes)
(crontab -l 2>/dev/null; echo "*/5 * * * * /root/health_check.sh") | crontab -

echo "✅ Health monitoring configured (5-minute intervals)"
```

### Setup Log Rotation
```bash
# Configure log rotation
cat > /etc/logrotate.d/trae-bot << 'EOF'
/root/ai-trading-sentinel/logs/*.log {
    daily
    rotate 30
    compress
    delaycompress
    missingok
    notifempty
    copytruncate
    create 644 root root
}
EOF

echo "✅ Log rotation configured (30-day retention)"
```
