```bash
cd /root

cd ai-trading-sentinel

# Clone repository (or create from local files)
echo "📥 Setting up AI Trading Sentinel..."
if [ ! -f "main.py" ]; then

#!/usr/bin/env python3
"""
AI Trading Sentinel - Main Application
VNC-Optimized Version
"""

import os
import sys
import time
import logging
from datetime import datetime

# Playwright imports (replacing selenium)
from playwright.sync_api import sync_playwright

# Other imports
try:
    import yaml
    import requests
    import flask
    from flask import Flask, jsonify
except ImportError as e:
    print(f"Missing dependency: {e}")
    sys.exit(1)

# Setup logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('/root/ai-trading-sentinel/logs/trae.log'),
        logging.StreamHandler()
    ]
)

logger = logging.getLogger(__name__)

class TradingBot:
    def __init__(self):
        self.running = False
        self.playwright = None
        self.browser = None
        self.page = None
        
    def start_browser(self):
        """Initialize Playwright browser"""
        try:
            self.playwright = sync_playwright().start()
            self.browser = self.playwright.chromium.launch(
                headless=False,  # Set to True for production
                args=['--no-sandbox', '--disable-dev-shm-usage']
            )
            self.page = self.browser.new_page()
            logger.info("Browser started successfully")
            return True
        except Exception as e:
            logger.error(f"Failed to start browser: {e}")
            return False
    
    def stop_browser(self):
        """Clean up browser resources"""
        try:
            if self.page:
                self.page.close()
            if self.browser:
                self.browser.close()
            if self.playwright:
                self.playwright.stop()
            logger.info("Browser stopped successfully")
        except Exception as e:
            logger.error(f"Error stopping browser: {e}")
    
    def run(self):
        """Main bot execution loop"""
        logger.info("AI Trading Sentinel starting...")
        self.running = True
        
        try:
            if not self.start_browser():
                return False
                
            while self.running:
                logger.info("Bot heartbeat - System operational")
                time.sleep(60)  # Check every minute
                
        except KeyboardInterrupt:
            logger.info("Shutdown signal received")
        except Exception as e:
            logger.error(f"Bot error: {e}")
        finally:
            self.stop_browser()
            self.running = False
            logger.info("AI Trading Sentinel stopped")

# Flask API for monitoring
app = Flask(__name__)

@app.route('/health')
def health_check():
    return jsonify({
        'status': 'healthy',
        'timestamp': datetime.now().isoformat(),
        'version': '1.0.0'
    })

@app.route('/status')
def bot_status():
    return jsonify({
        'bot_running': bot.running if 'bot' in globals() else False,
        'timestamp': datetime.now().isoformat()
    })

if __name__ == "__main__":
    # Create bot instance
    bot = TradingBot()
    
    # Start bot
    try:
        bot.run()
    except Exception as e:
        logger.error(f"Fatal error: {e}")
        sys.exit(1)
MAIN_EOF

#!/usr/bin/env python3
"""
AI Trading Sentinel Health Check
VNC Environment Optimized
"""

import requests
import subprocess
import sys
from datetime import datetime

def check_service_status():
    """Check systemd service status"""
    try:
        result = subprocess.run(['systemctl', 'is-active', 'trae-bot'], 
                              capture_output=True, text=True)
        return result.stdout.strip() == 'active'
    except:
        return False

def check_api_health():
    """Check Flask API health endpoint"""
    try:
        response = requests.get('http://localhost:5000/health', timeout=5)
        return response.status_code == 200
    except:
        return False

def check_log_activity():
    """Check if logs are being written"""
    try:
        result = subprocess.run(['tail', '-n', '1', '/root/ai-trading-sentinel/logs/trae.log'],
                              capture_output=True, text=True)
        return len(result.stdout.strip()) > 0
    except:
        return False

def main():
    print(f"Health Check - {datetime.now()}")
    print("=" * 40)
    
    service_ok = check_service_status()
    api_ok = check_api_health()
    logs_ok = check_log_activity()
    
    print(f"Service Status: {'✅ Active' if service_ok else '❌ Inactive'}")
    print(f"API Health: {'✅ Healthy' if api_ok else '❌ Unhealthy'}")
    print(f"Log Activity: {'✅ Active' if logs_ok else '❌ No Activity'}")
    
    overall_health = service_ok and api_ok and logs_ok
    print(f"Overall Status: {'✅ HEALTHY' if overall_health else '❌ UNHEALTHY'}")
    
    return 0 if overall_health else 1

if __name__ == "__main__":
    sys.exit(main())
HEALTH_EOF
```

```bash
#!/bin/bash
unset SESSION_MANAGER
unset DBUS_SESSION_BUS_ADDRESS
exec startxfce4
VNC_EOF

chmod +x ~/.vnc/xstartup
```

## VNC SECURITY CONSIDERATIONS

### 1. VNC Password Protection
```bash
# Set strong VNC password
vncpasswd

# Configure VNC to require password
echo "password" > ~/.vnc/passwd
chmod 600 ~/.vnc/passwd
```

### 2. VNC over SSH Tunnel (Optional)
```bash
# From local machine:
ssh -L 5901:localhost:5901 root@161.97.112.146
# Then connect VNC client to localhost:5901
```

### 3. Firewall Configuration
```bash
# Allow VNC port (if using direct connection)
ufw allow 5901/tcp

# Or restrict to specific IPs
ufw allow from YOUR_IP to any port 5901
```

## TROUBLESHOOTING VNC ISSUES

### VNC Connection Problems
```bash
# Restart VNC server
vncserver -kill :1
vncserver :1 -geometry 1920x1080

# Check VNC processes
ps aux | grep vnc

# Check VNC logs
cat ~/.vnc/*.log
```

### Desktop Environment Issues
```bash
# Install full desktop if minimal
sudo apt install ubuntu-desktop

# Reset VNC configuration
rm -rf ~/.vnc
vncserver :1
```
