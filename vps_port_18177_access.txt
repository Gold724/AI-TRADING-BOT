### Step 3: Execute Playwright Fix Script
```bash
# Create the fix script
cat > vps_playwright_fix.sh << 'EOF'
#!/bin/bash
set -e

echo "🔧 TRAE-BOT VPS Playwright Migration & Fix Script"
echo "================================================"

# Update system
apt update && apt upgrade -y

# Install Python and dependencies
apt install -y python3 python3-pip python3-venv git curl wget

# Install Node.js (for Playwright browsers)
curl -fsSL https://deb.nodesource.com/setup_18.x | bash -
apt install -y nodejs

# Create virtual environment
python3 -m venv /root/trae-venv
source /root/trae-venv/bin/activate

# Install Python packages
pip install --upgrade pip
pip install playwright pyyaml requests flask python-dotenv
pip install pandas numpy aiohttp websockets

# Install Playwright browsers
playwright install
playwright install-deps

# Fix main.py - replace selenium with playwright
if [ -f "main.py" ]; then
    echo "📝 Fixing main.py imports..."
    sed -i 's/from selenium/# from selenium (replaced with playwright)/g' main.py
    sed -i 's/import selenium/# import selenium (replaced with playwright)/g' main.py
    echo "✅ Selenium imports commented out"
fi

# Create logs directory
mkdir -p /root/AI-TRADING-BOT/logs
touch /root/AI-TRADING-BOT/logs/trae.log
chmod 755 /root/AI-TRADING-BOT/logs
chmod 644 /root/AI-TRADING-BOT/logs/trae.log

# Fix systemd service
cat > /etc/systemd/system/trae-bot.service << 'SERVICE_EOF'
[Unit]
Description=TRAE Trading Bot
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/root/AI-TRADING-BOT
Environment=PATH=/root/trae-venv/bin
ExecStart=/root/trae-venv/bin/python main.py
Restart=always
RestartSec=10
StandardOutput=append:/root/AI-TRADING-BOT/logs/trae.log
StandardError=append:/root/AI-TRADING-BOT/logs/trae.log

[Install]
WantedBy=multi-user.target
SERVICE_EOF

# Reload and start service
systemctl daemon-reload
systemctl enable trae-bot
systemctl start trae-bot

echo "✅ Deployment complete!"
echo "📊 Service status:"
systemctl status trae-bot --no-pager

echo "📋 Recent logs:"
tail -20 /root/AI-TRADING-BOT/logs/trae.log

echo "🎯 Management commands:"
echo "  systemctl status trae-bot    # Check status"
echo "  systemctl restart trae-bot   # Restart service"
echo "  tail -f /root/AI-TRADING-BOT/logs/trae.log  # Watch logs"
EOF

# Make executable and run
chmod +x vps_playwright_fix.sh
./vps_playwright_fix.sh
```
