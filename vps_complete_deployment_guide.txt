## DEPLOYMENT SCRIPT - Execute After Connection

### Step 1: Create Deployment Script
```bash
# Create the deployment script
cat > /root/deploy_trae_bot.sh << 'EOF'
#!/bin/bash
set -e

echo "=== AI Trading Sentinel - VPS Deployment ==="
echo "Starting deployment on Ubuntu 24.04..."

# Update system
echo "[1/10] Updating system packages..."
apt update && apt upgrade -y

# Install essential packages
echo "[2/10] Installing essential packages..."
apt install -y python3 python3-pip python3-venv nodejs npm git curl wget unzip

# Install Python packages
echo "[3/10] Installing Python packages..."
pip3 install --upgrade pip
pip3 install playwright pyyaml requests flask python-dotenv pandas numpy aiohttp websockets gunicorn

# Install Playwright browsers
echo "[4/10] Installing Playwright browsers..."
playwright install
playwright install-deps

# Navigate to bot directory
echo "[5/10] Setting up bot directory..."
cd /root
if [ ! -d "ai-trading-sentinel" ]; then
    echo "Cloning repository..."
    git clone https://github.com/YOUR_USERNAME/ai-trading-sentinel.git
fi
cd ai-trading-sentinel

# Create logs directory
echo "[6/10] Creating logs directory..."
mkdir -p logs
touch logs/trae.log
chmod 755 logs
chmod 644 logs/trae.log

# Fix main.py - Replace selenium with playwright
echo "[7/10] Fixing main.py imports..."
if [ -f "main.py" ]; then
    # Comment out selenium imports
    sed -i 's/^from selenium/#from selenium/g' main.py
    sed -i 's/^import selenium/#import selenium/g' main.py
    
    # Add playwright imports if not present
    if ! grep -q "from playwright" main.py; then
        echo "# Playwright imports (replacing selenium)" >> main.py
        echo "from playwright.async_api import async_playwright" >> main.py
        echo "from playwright.sync_api import sync_playwright" >> main.py
    fi
fi

# Create systemd service
echo "[8/10] Creating systemd service..."
cat > /etc/systemd/system/trae-bot.service << 'SERVICE_EOF'
[Unit]
Description=AI Trading Sentinel Bot
After=network.target
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/root/ai-trading-sentinel
Environment=PYTHONPATH=/root/ai-trading-sentinel
Environment=DISPLAY=:0
ExecStartPre=/bin/bash -c 'echo "Starting TRAE Bot at $(date)" >> /root/ai-trading-sentinel/logs/trae.log'
ExecStart=/usr/bin/python3 /root/ai-trading-sentinel/main.py
Restart=always
RestartSec=10
StandardOutput=append:/root/ai-trading-sentinel/logs/trae.log
StandardError=append:/root/ai-trading-sentinel/logs/trae.log

[Install]
WantedBy=multi-user.target
SERVICE_EOF

# Reload systemd and enable service
echo "[9/10] Configuring systemd service..."
systemctl daemon-reload
systemctl enable trae-bot.service

# Test Python imports
echo "[10/10] Testing Python imports..."
python3 -c "import playwright; import yaml; import requests; import flask; print('✅ All imports successful!')"

echo "\n=== DEPLOYMENT COMPLETE ==="
echo "✅ System updated and packages installed"
echo "✅ Playwright browsers installed"
echo "✅ Logs directory created"
echo "✅ main.py fixed (selenium → playwright)"
echo "✅ Systemd service configured"
echo "✅ Python imports verified"

echo "\n=== NEXT STEPS ==="
echo "1. Configure .env file with your credentials"
echo "2. Start the service: systemctl start trae-bot"
echo "3. Check status: systemctl status trae-bot"
echo "4. View logs: tail -f logs/trae.log"

EOF

# Make script executable
chmod +x /root/deploy_trae_bot.sh
```

### Setup SSH Keys (Recommended)
```bash
# On your local machine, generate SSH key
ssh-keygen -t rsa -b 4096 -f ~/.ssh/contabo_vps

# Copy public key to VPS
ssh-copy-id -i ~/.ssh/contabo_vps.pub -p 18177 root@161.97.112.146

# Then connect with key
ssh -i ~/.ssh/contabo_vps -p 18177 root@161.97.112.146
```


## 24/7 MONITORING SETUP

### Health Check Script
```bash
cat > /root/health_check.sh << 'EOF'
#!/bin/bash
if ! systemctl is-active --quiet trae-bot; then
    echo "$(date): TRAE Bot service is down, restarting..." >> /root/ai-trading-sentinel/logs/health.log
    systemctl restart trae-bot
    sleep 30
    if systemctl is-active --quiet trae-bot; then
        echo "$(date): TRAE Bot service restarted successfully" >> /root/ai-trading-sentinel/logs/health.log
    else
        echo "$(date): CRITICAL: TRAE Bot service failed to restart" >> /root/ai-trading-sentinel/logs/health.log
    fi
fi
EOF

chmod +x /root/health_check.sh

# Add to crontab (check every 5 minutes)
(crontab -l 2>/dev/null; echo "*/5 * * * * /root/health_check.sh") | crontab -
```
