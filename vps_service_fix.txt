# VPS Service Fix - Copy and Paste This Block to Fix systemd Service
# This fixes the exit code 203/EXEC error

# Stop the failing service
systemctl stop trae-bot

# Check current working directory and files
echo "Current directory: $(pwd)"
ls -la

# Verify Python and main.py exist
echo "Checking Python virtual environment..."
ls -la venv/bin/
echo "Checking main.py..."
ls -la main.py

# Create corrected systemd service
echo "Creating corrected systemd service..."
cat > /etc/systemd/system/trae-bot.service << 'EOF'
[Unit]
Description=AI Trading Sentinel Bot
After=network.target
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/root/AI-TRADING-BOT
Environment=PATH=/root/AI-TRADING-BOT/venv/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
Environment=PYTHONPATH=/root/AI-TRADING-BOT
Environment=DISPLAY=:99
ExecStartPre=/bin/bash -c 'pkill Xvfb || true'
ExecStartPre=/usr/bin/Xvfb :99 -screen 0 1024x768x24 -nolisten tcp -ac &
ExecStart=/bin/bash -c 'cd /root/AI-TRADING-BOT && source venv/bin/activate && python main.py'
Restart=always
RestartSec=10
KillMode=mixed
TimeoutStopSec=30
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
EOF

# Reload systemd daemon
systemctl daemon-reload

# Test the bot manually first
echo "Testing bot manually..."
source venv/bin/activate
python -c "print('Python environment OK')"
python -c "import sys; print('Python path:', sys.path)"

# Check if main.py exists and is executable
if [ -f "main.py" ]; then
    echo "✅ main.py found"
    python -c "import ast; ast.parse(open('main.py').read()); print('✅ main.py syntax OK')"
else
    echo "❌ main.py not found - creating basic main.py"
    cat > main.py << 'MAINEOF'
#!/usr/bin/env python3
"""
AI Trading Sentinel - Main Entry Point
"""

import os
import sys
import time
import logging
from datetime import datetime

# Setup logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('/root/AI-TRADING-BOT/bot.log'),
        logging.StreamHandler(sys.stdout)
    ]
)

logger = logging.getLogger(__name__)

def main():
    """Main trading bot entry point"""
    logger.info("🚀 AI Trading Sentinel Starting...")
    logger.info(f"Working Directory: {os.getcwd()}")
    logger.info(f"Python Path: {sys.path}")
    
    # Load environment variables
    try:
        from dotenv import load_dotenv
        load_dotenv()
        logger.info("✅ Environment variables loaded")
    except ImportError:
        logger.warning("⚠️ python-dotenv not installed, using system env vars")
    
    # Check simulation mode
    simulation_mode = os.getenv('SIMULATION_MODE', 'true').lower() == 'true'
    logger.info(f"📊 Simulation Mode: {simulation_mode}")
    
    # Main trading loop
    try:
        while True:
            logger.info(f"💹 Trading cycle at {datetime.now()}")
            
            # Add your trading logic here
            # For now, just log that we're running
            logger.info("🔄 Bot is running and monitoring markets...")
            
            # Sleep for 60 seconds between cycles
            time.sleep(60)
            
    except KeyboardInterrupt:
        logger.info("🛑 Bot stopped by user")
    except Exception as e:
        logger.error(f"❌ Bot error: {e}")
        raise

if __name__ == "__main__":
    main()
MAINEOF
    chmod +x main.py
fi

# Test the service configuration
echo "Testing service configuration..."
systemctl status trae-bot --no-pager

# Start the service
echo "Starting trae-bot service..."
systemctl start trae-bot

# Wait a moment and check status
sleep 5
echo "Service status after start:"
systemctl status trae-bot --no-pager

# Show recent logs
echo "Recent service logs:"
journalctl -u trae-bot -n 20 --no-pager

# Enable auto-start
systemctl enable trae-bot

echo ""
echo "🎉 Service Fix Complete!"
echo "📊 Monitor with: journalctl -u trae-bot -f"

echo "🛑 Stop with: systemctl stop trae-bot"
echo ""
