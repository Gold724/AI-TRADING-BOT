#!/bin/bash
# VPS Final Fix - Resolve remaining issues
# Copy and paste this script into your VPS terminal

echo "🔧 FINAL FIX - Resolving Remaining Issues"
echo "========================================="
echo "Timestamp: $(date)"
echo ""

# Navigate to correct directory
cd /root/AI-TRADING-BOT
echo "📁 Working in: $(pwd)"
echo ""

# Stop the service first
echo "🛑 Stopping service..."
systemctl stop trae-bot
echo ""

# Fix Issue 1: Create missing logs directory
echo "📁 Creating logs directory..."
mkdir -p logs
touch logs/trae.log
chmod 755 logs
chmod 644 logs/trae.log
echo "✅ Logs directory created: $(ls -la logs/)"
echo ""

# Fix Issue 2: Correct systemd service syntax
echo "⚙️ Fixing systemd service configuration..."
cat > /etc/systemd/system/trae-bot.service << 'EOF'
[Unit]
Description=AI Trading Sentinel Bot
After=network.target
Wants=network-online.target

[Service]
Type=simple
User=root
WorkingDirectory=/root/AI-TRADING-BOT
Environment=PATH=/root/AI-TRADING-BOT/venv/bin:/usr/local/bin:/usr/bin:/bin
ExecStartPre=/bin/bash -c 'cd /root/AI-TRADING-BOT && source venv/bin/activate && python -c "print(\"Environment check OK\")"'
ExecStart=/bin/bash -c 'cd /root/AI-TRADING-BOT && source venv/bin/activate && exec python main.py'
Restart=always
RestartSec=10
StartLimitIntervalSec=300
StartLimitBurst=5
StandardOutput=journal
StandardError=journal
SyslogIdentifier=trae-bot

[Install]
WantedBy=multi-user.target
EOF
echo "✅ Systemd service configuration fixed"
echo ""

# Activate virtual environment for testing
echo "🐍 Activating virtual environment..."
source venv/bin/activate
echo "Python version: $(python --version)"
echo ""

# Test main.py execution again
echo "🧪 Testing main.py with logs directory..."
timeout 3s python main.py 2>&1 | head -10
exit_code=$?
if [ $exit_code -eq 124 ]; then
    echo "✅ main.py started successfully (timed out as expected)"
elif [ $exit_code -eq 0 ]; then
    echo "✅ main.py executed successfully"
else
    echo "❌ main.py still failing, checking error..."
    python main.py 2>&1 | head -5
fi
echo ""

# Reload systemd and start service
echo "🔄 Reloading systemd daemon..."
systemctl daemon-reload
echo ""

echo "🚀 Starting trae-bot service..."
systemctl start trae-bot
sleep 5
echo ""

echo "📊 Service Status Check:"
systemctl status trae-bot --no-pager -l
echo ""

echo "📋 Recent Service Logs:"
journalctl -u trae-bot --no-pager -n 15
echo ""

# Final status summary
service_status=$(systemctl is-active trae-bot)
service_enabled=$(systemctl is-enabled trae-bot)

echo "🎯 FINAL STATUS SUMMARY:"
echo "========================"
echo "Service Status: $service_status"
echo "Service Enabled: $service_enabled"
echo "Logs Directory: $(ls -la logs/ 2>/dev/null || echo 'Missing')"
echo ""

if [ "$service_status" = "active" ]; then
    echo "🎉 SUCCESS! Service is now running properly!"
    echo "✅ All issues resolved:"
    echo "   - Logs directory created"
    echo "   - Systemd service syntax fixed"
    echo "   - Service is active and running"
else
    echo "⚠️ Service still not active. Checking for remaining issues..."
    echo "Recent error logs:"
    journalctl -u trae-bot --no-pager -n 5 | grep -E "(ERROR|Failed|Error)"
fi

echo ""
echo "📋 Management Commands:"
echo "- Check status: systemctl status trae-bot"
echo "- View live logs: journalctl -u trae-bot -f"
echo "- Restart: systemctl restart trae-bot"
echo "- Stop: systemctl stop trae-bot"
echo "- Manual test: cd /root/AI-TRADING-BOT && source venv/bin/activate && python main.py"
echo ""
echo "🔧 If still failing, run manual test to see specific errors."
